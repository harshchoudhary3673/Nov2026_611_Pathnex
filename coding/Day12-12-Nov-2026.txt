# ansible - create docker container for multiple applications
#------------------------------------------------------------------------------------------------------------------

- name: steup docker for multiple applications
  host: all
  become: yes
  tasks:
    - name: pull web app image
      docker_image:
        name: my-web-app
        source: pull

    - name: pull database app image
      docker_image:
        name: my-database-app
        source: pull

    - name: run web app container
      docker_container:
        name: web-app
        image: my-web-app
        state: started
        published_ports:
          - "8080:80"

    - name: run database app container
      docker_container:
        name: database-app
        image: my-database-app
        state: started
        published_ports:
          - "3306:3306"

#Kubernetes - deploying statefulset
#------------------------------------------------------------------------------------------------------------------

apiVersion: apps/v1
kind: statefulset
metadata:
  name: pathnex-db 
spec:
  servicename: "pathnex-db"
  replicas: 3
  selector: 
    matchlabels:
      app: pathnex-db
  template:
    metadata:
      labels:
        app: pathnex-db
    spec:
      containers:
        - name: db
          image: mysql:5.7
          volumemounts:
            - Mountpath: /var/lib/mysql
              name: mysql-storage
  volumeClaimtemplates:
    - metadata:
        name: mysql-storage
      spec:
        accessmodes: ["ReadWriteOnce"]
        resources:
          requests:
            storage: 1Gi

#jenkinsfile - build and deploy to kubernetes
#------------------------------------------------------------------------------------------------------------------

pipeline {
    agent any
    environment {
        IMAGE_NAME = 'pathnex-web-app'
    }
    stages{
        stage('build docker image')
            steps {
                script {
                    docker.build("${env.IMAGE_NAME}")
                }
            }
        }
        stage('push docker Image') {
            steps {
                script {
                    docker.withRegistry('https://docker.io', 'docker-credentials') {
                        docker.image("${env.IMAGE_NAME}").push('latest')
                    }
                }
            }
        }
        stage('deploy to kubernetes') {
            steps {
                script {
                    docker.image("${env.IMAGE_NAME}").push('latest')
                }
            }
        }
  }

#GitLab CI/CD - Build , Push, and Deploy
#------------------------------------------------------------------------------------------------------------------

Stages:
  - Build
  - Push
  - deploy

Build:
  stage: Build
  Script:
    - docker build -t pathnex-web-app .

push:
  stage: push
  script: 
    - docker login -u "$CI_REGISTRY_USER" -p "$CI_REGISTRY_PASSWORD"
    - docker push pathnex-web-app

deploy:
  stage: deploy
  script:
    - kubectl apply -f kubernetes/deployment.yaml
