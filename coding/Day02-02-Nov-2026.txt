# ansible - install and enble nginx
- name: instal and enable nginx on pathnex
  host: all
  become: yes

  tasks:
    - name: install nginx
      yum:
        name: nginx
        state: present	


    - name: enable nginx
   	service:
  	  name: nginx
	  state: started
	  enabled: yes

# kubernetes - Deployment with 2 replica

apiVersion: app/v1
kind: Deployment
metadata:
  name: pathnex-deployment
spec:
  replicas: 2
  selector:
    matchlabels:
      app: pathnex-app
  template:
    metadata:
      labels: 
	app: pathnex-app
    spec:
      containers:
	- name: app
	  image: nginx
   	  ports:
	    - containerports: 80
