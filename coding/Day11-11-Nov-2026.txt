#------------------------------------------------------------------------------------------------------------------
# docker integration with jenkins and Gitlab
#------------------------------------------------------------------------------------------------------------------

#------------------------------------------------------------------------------------------------------------------
# Ansible - setup docker container for nginx
#------------------------------------------------------------------------------------------------------------------
- name: setup docker container for nginx
  hosts: all
  become: yes
  tasks:
    - name: pull nginx image
      docker_image: 
        name: nginx
        source: pull
    - name: run nginx container
      docker_container:
        name: nginx-container
        image: nginx
        state: started
        published_ports:
          - "8080:80"

#-----------------------------------------------------------------------------------------------------------------
#Terraform -EC2 with auto scaling group
#-----------------------------------------------------------------------------------------------------------------
resource "aws_launch_configuration" "example" {
  name                        = "english-config"
  Image_id                    = "ami-0abcd1234abcd1234"
  instance_type               = "t3.medium"
}

resource "aws_autoscaling_group" "example" {
  desired_capacity            =  2
  max_size                    =  3
  min_size                    =  1
  vpc_zone_identifier         =  ["subnet-12345678"]
  launch_configuration        =  aws_launch_configuration.example.id
}

#-----------------------------------------------------------------------------------------------------------------
#kubernetes - Horizontal pod  autoscaler (HPA)
#-----------------------------------------------------------------------------------------------------------------
apiVersion: app/v1
kind: deployment
metadata:
  name: pathnex-deployment
spec:
  replica: 1
  selector: 
    matchlabels:
      app: pathnex-app
  template:
    metadata:
      labels:
        app: pathnex-app
    spec:
      containers:
        - name: nginx
          image: nginx
          ports: 
            - containerport: 80
#-----------------------------------------------------------------------------------------------------------------
apiVersion: autoscaling/v2
kind: Horizontalpodautoscaler
metadata:
  name: Pathnex-HPA
spec:
  scaletragetref:
    apiVersion: app/v1
    kind: deployment
    name: Pathnex-deployment
  Minreplica: 1
  maxreplica: 10
  targetCPUutilizationpercentage: 50%

#-----------------------------------------------------------------------------------------------------------------
#jenkins - deploy to kubernetes
#-----------------------------------------------------------------------------------------------------------------
pipeline {
    agent any
    stages {
        stage('deploy to kubernetes') {
            steps {
                script {
                    sh 'kubectl apply -f deployment.yaml'
                }
            }
        }
    }
}

#-----------------------------------------------------------------------------------------------------------------
# GitLab CI/CD - Deploy to kubernetes
#-----------------------------------------------------------------------------------------------------------------

stages:
  - deploy

deploy:
  stage: deploy
  script:
    - kubectl apply -f kubernetes/deployment.yaml

from ubuntu: 22.04
