# Jenkinsfile - with git integration and testing


pipeline {
    agents any
    stages {
        stage ('Clone repositoy') {
            steps {
                git 'https://github.com/pathnex/sample-repo.git'
            
            }
        }
        stage('Build') {
            steps {
                echo 'Building project from git repositoy'
            }
        }
        stage('test') {
            steps {
                echo 'Running unit tests'
            }
        }
        stage('Deploy') {
            steps {
                echo 'Deploying application'
            
            }
        }

# GITLAB CI/CD - with docker Build

  - Deploy

# Ansible - create user and permission

- name: create a new user
  hosts: all
  become: yes
  tasks:
    - name: create user
      user: 
        name: pathnex
        state: present
        shell: /bin/bash

    - name: set permissions for the user 
      file:
        path: home/pathnex
        owner: pathnex
        group: pathnex
        mode: '0755'
